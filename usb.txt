#include <AT89S53.H>
#include <stdio.h>
#include <math.h>

unsigned long Z1, timer1, /*dt, */Time, t1, t, /*Time1,*/ dose, Time_z1, t_du, t_du1, t_du2, t_du_s;
signed long timer_ds1621, timer_ds1621_in;
unsigned int i;
unsigned char temp_ds1621, temp_ds1621_2, in, i1, t2, inport, lpt[16], Z2, req;
bit flag, flag_lev, flag_temperature, flag_z1, flag_serial, flag_ds1621_dis, flag_du;

#define Delay 20 		
#define serial1 53 		//номер блока		
#define serial2 7		//номер проги
#define serial3 10		// дата блока  месяц
#define serial4 6		// дата блока  год +2000
#define serial5 0		// дата начала лизинга день
#define serial6 0		// дата начала лизинга  месяц
#define serial7 0		// дата начала лизинга  год +2000
#define serial8 0		// дата окончания лизинга  день
#define serial9 0		// дата окончания лизинга месяц
#define serial10 0		// дата окончания лизинга год +2000
#define serial11 0
#define serial12 0

void sendbit(bit snbit)
{
	for (i = 0; i < Delay; i++);
	P1_5 = snbit;
	for (i = 0; i < Delay; i++);
	P1_7 = 1;

	for (i = 0; i < Delay; i++);
	P1_7 = 0;
	for (i = 0; i < Delay; i++);
}

void inbit(void)
{
	for (i = 0; i < Delay; i++);
	in = P1_5;
	for (i = 0; i < Delay; i++);
	P1_7 = 1;
	for (i = 0; i < Delay; i++);
	P1_7 = 0;
	temp_ds1621 = temp_ds1621 << 1;
	temp_ds1621 += in;
	for (i = 0; i < Delay; i++);
}

void inbit2(void)
{
	for (i = 0; i < Delay; i++);
	in = P1_5;
	for (i = 0; i < Delay; i++);
	P1_7 = 1;
	for (i = 0; i < Delay; i++);
	P1_7 = 0;
	temp_ds1621_2 = temp_ds1621_2 << 1;
	temp_ds1621_2 += in;
	for (i = 0; i < Delay; i++);
}

void int0(void) interrupt 0
{
	Z1 = Z1 + 1; // счетчик импульсов
	flag_z1 = 1;
	Time_z1 = t;
	if (Z1 >= dose)
		P3_4 = 1; // закрытие клапана
}

void timer0(void) interrupt 1
{
	t = t + 1;   // дискретный t  50 (100) миллисекунд
	t2 = t2 + 1;
	if (t2 == 9)
		P3_5 = 1;  // 1.1KHz
	if (t2 == 18)
	{
		P3_5 = 0;
		t2 = 0;
	}
}

void main(void)
{
	//unsigned long  s1;
	unsigned char lpt_in[10], j1, j /*,s[8]*/;

	P3_4 = 1; // закрытие клапана
	t2 = 0;
	t = 0;
	t1 = 0;
	Z1 = 0;
	Z2 = 0;
	temp_ds1621 = 0;
	P3_0 = 1;
	P3_1 = 1;
	P3_2 = 1;
	P1_1 = 1;
	P1_4 = 1;
	flag_z1 = 0;
	timer_ds1621 = -199000;
	timer_ds1621_in = -100000;
	temp_ds1621_2 = 0;
	flag_ds1621_dis = 0;
	t_du = 0;
	t_du2 = 0;
	t_du_s = 0;
	t_du1 = 0;
	flag_du = 0;

	P1_5 = 1;   //sda
	P1_7 = 1;   //scl
	for (i = 0; i < Delay; i++);
	P1_5 = 0;     // Start
	for (i = 0; i < Delay; i++);
	P1_7 = 0;   //scl

	sendbit(1);
	sendbit(0);
	sendbit(0);
	sendbit(1);
	sendbit(0);
	sendbit(0);
	sendbit(0);
	sendbit(0);

	P1_7 = 1;		/// Ack 1621
	for (i = 0; i < Delay; i++);
	P1_7 = 0;

	sendbit(1);		//EE
	sendbit(1);
	sendbit(1);
	sendbit(0);
	sendbit(1);
	sendbit(1);
	sendbit(1);
	sendbit(0);

	P1_7 = 1;		/// Ack 1621
	for (i = 0; i < Delay; i++);
	P1_7 = 0;
	for (i = 0; i < Delay; i++);

	P1_5 = 0;
	for (i = 0; i < Delay; i++);
	P1_7 = 1;  //Stop
	for (i = 0; i < Delay; i++);
	P1_5 = 1;

	////////////////////////// инициализация индикатора
	for (i = 0; i < 40000; i++);
	/////////////////////////////////
	/* setup serial port control hardware (9600 BAUD @24MHZ) */
	PCON = 0x80;    /*smod=1 удвоение скорости */
	SCON = 0xd2;    /* SCON  11010010 */
	TMOD = 0x22;    /* TMOD  00100010 */
	TCON = 0x50;    /* TCON  01010000 */
	TH1  = 0xf3;    /* TH1 */
	TH0  = 0x9c; //38;    /* TH0 */

	EX1 = 0;
	EX0 = 1;                 /* enable int0 interrupt */
	ET0 = 1;                 /* enable timer 0 interrupt */
	EA = 1;                  /* global interrupt enable */
	IT0 = 1;
	/////////////////// Сторожевой таймер

	WMCON = 0x01;
	WMCON = WMCON | 0x20;     /* 011 =  128ms Timeout */
	WMCON = WMCON | 0x40;
	// WMCON = WMCON | 0x80;
	//////////////////
	while (1)
	{
		///////////////////////////// сигнализатор ДУ
		t_du1 = t_du;
		t_du  = t;
		t_du2 = t_du - t_du1;
		if (P1_3 == 0)
			t_du_s = 0;
		flag_du = 0;
		if (P1_3 == 1)
		{
			t_du_s = t_du_s + t_du2;
			if ((t_du_s > 120) || (flag_du == 1)) // 6 миллисекунд
			{
				P3_4 = 1; // закрытие клапана
				flag_z1 = 1;
				t_du_s = 0;
			}
			flag_du = 1;
		}
		///////////////////////// LPT
		P1_1 = 1;
		if (P3_3 == 0)      /// fifo не пусто   RXF
		{
			for (j1 = 0; j1 < 5; j1++)   ///чтение кадра из fifo 5 байт
			{
				P1_1 = 0;
				for (i = 0; i < 5; i++);
				lpt_in[j1] = P2;
				for (i = 0; i < 5; i++);
				P1_1 = 1;
				for (i = 0; i < 5; i++);
			}

			if (lpt_in[0] == 0x55)
			{
				t = 0;
				timer_ds1621 = -199000;
				timer_ds1621_in = -100000;
				Z1 = 0;
				dose = lpt_in[1] + (lpt_in[2] * 256) + (lpt_in[3] * 65536) + (lpt_in[4] * 16777216);
				P3_4 = 0; // открытие клапана
				flag_z1 = 1;
				flag_serial = 1;
			}

			if (lpt_in[0] == 0xaa)
			{
				P3_4 = 1; // закрытие клапана
				flag_z1 = 1;
			}
		}

		if (flag_z1 == 1)
		{
			Time = t;

			lpt[0] = 0x55;

			if (flag_serial == 1)
			{
				lpt[1] = serial1;
				lpt[2] = serial2;
				lpt[3] = serial3;
				lpt[4] = serial4;
				lpt[5] = serial5;
				lpt[6] = serial6;
				lpt[7] = serial7;
				lpt[8] = serial8;

				lpt[9] = serial9;
				lpt[10] = serial10;
				lpt[11] = serial11;
				lpt[12] = serial12;

				flag_serial = 0;
			}
			else
			{
				lpt[1] = Time;		///// время
				lpt[2] = ((Time >> 8));
				lpt[3] = ((Time >> 16));
				lpt[4] = ((Time >> 24));
				lpt[5] = Time_z1; 		///// время прихода последнего импульса
				lpt[6] = ((Time_z1 >> 8));
				lpt[7] = ((Time_z1 >> 16));
				lpt[8] = ((Time_z1 >> 24));
				lpt[9] = (Z1);		////импульсы (0.1литр)
				lpt[10] = ((Z1 >> 8));
				lpt[11] = ((Z1 >> 16));
				lpt[12] = ((Z1 >> 24));
			}
			lpt[13] = temp_ds1621;		///  температура

			lpt[14] = 0;
			lpt[14] = P3_4;		/// состояние клапана
			if (P1_3 == 0x01)		/// состояние ДО
				lpt[14] = lpt[14] | 0x02;   /// ДО
			if (temp_ds1621_2 == 0xff)		///нет датчика температуры
				lpt[14] = lpt[14] | 0x04;

			if (temp_ds1621_2 == 0x80)		///1/2градуса   темп датчика
				lpt[14] = lpt[14] | 0x08;
			lpt[15] = 0;		///// crc
			for (i = 0; i < 15; i++)
				lpt[15] += lpt[i];
			if (P1_2 == 0)
			{
				timer1 = t;
				for (j = 0; j < 16; j++)   ///запись кадра в fifo 16байт
				{
					P2 = lpt[j];
					for (i = 0; i < 10; i++);
					P1_4 = 0;
					for (i = 0; i < 10; i++);
					P1_4 = 1;
				}
			}
			if (P1_2 == 1)
			{
				if ((t - timer1) > 20000)   //таймаут 1000 миллисекунд
					P3_4 = 1; // закрытие клапана
			}
		}
		P2 = 0xff;
		flag_z1 = 0;
		////////////////////////// DS 1621
		if ((t - timer_ds1621) > 200000)   //10 секунд
		{
			timer_ds1621 = t;
			P1_5 = 1;   //sda
			P1_7 = 1;   //scl
			for (i = 0; i < Delay; i++);
			P1_5 = 0;     // Start
			for (i = 0; i < Delay; i++);
			P1_7 = 0;   //scl

			sendbit(1);
			sendbit(0);
			sendbit(0);
			sendbit(1);
			sendbit(0);
			sendbit(0);
			sendbit(0);
			sendbit(0);

			P1_7 = 1;		/// Ack 1621
			for (i = 0; i < Delay; i++);
			P1_7 = 0;

			sendbit(1);   //AA
			sendbit(0);
			sendbit(1);
			sendbit(0);
			sendbit(1);
			sendbit(0);
			sendbit(1);
			sendbit(0);

			P1_7 = 1;		/// Ack 1621
			for (i = 0; i < Delay; i++);
			P1_7 = 0;
			for (i = 0; i < Delay; i++);

			P1_5 = 1;
			for (i = 0; i < Delay; i++);
			P1_7 = 1;   //sda
			for (i = 0; i < Delay; i++);

			P1_5 = 0;     // Start

			for (i = 0; i < Delay; i++);
			P1_7 = 0;   //scl

			sendbit(1);
			sendbit(0);
			sendbit(0);
			sendbit(1);
			sendbit(0);
			sendbit(0);
			sendbit(0);
			sendbit(1);

			P1_7 = 1;		/// Ack 1621
			for (i = 0; i < Delay; i++);
			P1_7 = 0;

			temp_ds1621 = 0;
			for (i1 = 0; i1 < 8; i1++)
				inbit();

			for (i = 0; i < Delay; i++);
			P1_5 = 0;
			for (i = 0; i < Delay; i++);
			P1_7 = 1;
			for (i = 0; i < Delay; i++);
			P1_7 = 0;
			P1_5 = 1;

			temp_ds1621_2 = 0;
			for (i1 = 0; i1 < 8; i1++)
				inbit2();

			P1_5 = 1;
			for (i = 0; i < Delay; i++);
			P1_7 = 1;		/// NAck MK
			for (i = 0; i < Delay; i++);
			P1_7 = 0;
			for (i = 0; i < Delay; i++);

			P1_5 = 0;
			for (i = 0; i < Delay; i++);
			P1_7 = 1;  //Stop
			for (i = 0; i < Delay; i++);
			P1_5 = 1;
		}

		if ((t - timer_ds1621_in) > 200000)   //10 секунд
		{
			timer_ds1621_in = t;

			P1_5 = 1;   //sda
			P1_7 = 1;   //scl
			for (i = 0; i < Delay; i++);

			P1_5 = 0;     // Start

			for (i = 0; i < Delay; i++);
			P1_7 = 0;   //scl

			sendbit(1);
			sendbit(0);
			sendbit(0);
			sendbit(1);
			sendbit(0);
			sendbit(0);
			sendbit(0);
			sendbit(0);

			P1_7 = 1;		/// Ack 1621
			for (i = 0; i < Delay; i++);
			P1_7 = 0;

			sendbit(1);   //EE
			sendbit(1);
			sendbit(1);
			sendbit(0);
			sendbit(1);
			sendbit(1);
			sendbit(1);
			sendbit(0);

			P1_7 = 1;		/// Ack 1621
			for (i = 0; i < Delay; i++);
			P1_7 = 0;
			for (i = 0; i < Delay; i++);

			P1_5 = 0;
			for (i = 0; i < Delay; i++);
			P1_7 = 1;  //Stop
			for (i = 0; i < Delay; i++);
			P1_5 = 1;
		}
		///

		WMCON = WMCON | 0x02;  // обнуление ст. таймера
	}
}
